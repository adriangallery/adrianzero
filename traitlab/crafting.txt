pragma solidity ^0.8.20;

import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/security/ReentrancyGuard.sol";

/* ========= Interface mínima de AdrianTraitsCore ========= */
interface IAdrianTraitsCore {
    function balanceOf(address account, uint256 id) external view returns (uint256);
    function mintFromExtension(address to, uint256 id, uint256 amount) external;
    function burnFromExtension(address from, uint256 id, uint256 amount) external;
    function authorizedExtensions(address extension) external view returns (bool);
    function exists(uint256 tokenId) external view returns (bool);
}

/**
 * @title AdrianCrafting
 * @notice Crafteo para ERC1155 (AdrianTraitsCore):
 *         - Recetas ESPECÍFICAS: quemar IDs/cantidades exactas -> mintear salida.
 *         - Recetas GENERALES por CANTIDAD: quemar N unidades de cualquier combinación -> mintear salida.
 * @dev Requiere autorización como extensión en AdrianTraitsCore.
 */
contract AdrianCrafting is Ownable, ReentrancyGuard {
    IAdrianTraitsCore public traitsCore;

    /* ============ Constructor ============ */
    constructor(address traitsCore_, address initialOwner) Ownable(initialOwner) {
        require(traitsCore_ != address(0), "Core addr required");
        traitsCore = IAdrianTraitsCore(traitsCore_);
    }

    /* ============ Modificadores ============ */

    modifier onlyWhenAuthorized() {
        require(traitsCore.authorizedExtensions(address(this)), "Crafting not authorized in Core");
        _;
    }

    /* ============ Modelos de Receta ============ */

    struct SpecificRecipe {
        bool active;
        uint256[] burnIds;
        uint256[] burnAmounts; // paralela a burnIds
        uint256 outId;
        uint256 outAmount; // normalmente 1
    }

    struct AnyRecipe {
        bool active;
        uint256 burnTotal; // p.ej. 7
        uint256 outId;
        uint256 outAmount; // normalmente 1
    }

    mapping(uint256 => SpecificRecipe) public specificRecipes; // recipeId => receta
    mapping(uint256 => AnyRecipe) public anyRecipes;           // recipeId => receta

    /* ============ Eventos ============ */

    event SpecificRecipeSet(
        uint256 indexed recipeId,
        bool active,
        uint256[] burnIds,
        uint256[] burnAmounts,
        uint256 outId,
        uint256 outAmount
    );

    event AnyRecipeSet(
        uint256 indexed recipeId,
        bool active,
        uint256 burnTotal,
        uint256 outId,
        uint256 outAmount
    );

    event CraftedSpecific(address indexed user, uint256 indexed recipeId, uint256 outId, uint256 outAmount);
    event CraftedAny(address indexed user, uint256 indexed recipeId, uint256 outId, uint256 outAmount, uint256 burnedTotal);

    /* ============ Setters (owner) ============ */

    /**
     * @notice Define/actualiza una receta específica.
     * @param recipeId        Identificador de la receta.
     * @param active_         Activar/desactivar.
     * @param burnIds_        IDs a quemar (p.ej. [1,43,72]).
     * @param burnAmounts_    Cantidades por ID (paralela a burnIds_).
     * @param outId_          ID a mintear (p.ej. 827).
     * @param outAmount_      Cantidad a mintear (normalmente 1).
     */
    function setSpecificRecipe(
        uint256 recipeId,
        bool active_,
        uint256[] calldata burnIds_,
        uint256[] calldata burnAmounts_,
        uint256 outId_,
        uint256 outAmount_
    ) external onlyOwner {
        require(burnIds_.length > 0, "Empty burnIds");
        require(burnIds_.length == burnAmounts_.length, "Length mismatch");
        require(outId_ != 0, "outId=0?");
        require(outAmount_ > 0, "outAmount=0");

        SpecificRecipe storage R = specificRecipes[recipeId];
        R.active = active_;
        R.outId = outId_;
        R.outAmount = outAmount_;

        delete R.burnIds;
        delete R.burnAmounts;
        for (uint256 i = 0; i < burnIds_.length; i++) {
            R.burnIds.push(burnIds_[i]);
            R.burnAmounts.push(burnAmounts_[i]);
        }

        emit SpecificRecipeSet(recipeId, active_, R.burnIds, R.burnAmounts, outId_, outAmount_);
    }

    /**
     * @notice Define/actualiza una receta general por cantidad.
     * @param recipeId    Identificador de la receta.
     * @param active_     Activar/desactivar.
     * @param burnTotal_  Unidades totales a quemar (p.ej. 7).
     * @param outId_      ID a mintear (p.ej. 538).
     * @param outAmount_  Cantidad a mintear (normalmente 1).
     */
    function setAnyRecipe(
        uint256 recipeId,
        bool active_,
        uint256 burnTotal_,
        uint256 outId_,
        uint256 outAmount_
    ) external onlyOwner {
        require(burnTotal_ > 0, "burnTotal=0");
        require(outId_ != 0, "outId=0?");
        require(outAmount_ > 0, "outAmount=0");

        anyRecipes[recipeId] = AnyRecipe({
            active: active_,
            burnTotal: burnTotal_,
            outId: outId_,
            outAmount: outAmount_
        });

        emit AnyRecipeSet(recipeId, active_, burnTotal_, outId_, outAmount_);
    }

    /* ============ Crafting ============ */

    /**
     * @notice Ejecuta una receta específica: quema los IDs/cantidades exactos y mintea la salida.
     */
    function craftSpecific(uint256 recipeId) external nonReentrant onlyWhenAuthorized {
        SpecificRecipe storage R = specificRecipes[recipeId];
        require(R.active, "Recipe inactive");

        for (uint256 i = 0; i < R.burnIds.length; i++) {
            uint256 id = R.burnIds[i];
            uint256 amt = R.burnAmounts[i];
            require(amt > 0, "Burn amt=0");
            require(traitsCore.balanceOf(msg.sender, id) >= amt, "Insufficient balance for an id");
        }

        for (uint256 i = 0; i < R.burnIds.length; i++) {
            traitsCore.burnFromExtension(msg.sender, R.burnIds[i], R.burnAmounts[i]);
        }

        traitsCore.mintFromExtension(msg.sender, R.outId, R.outAmount);

        emit CraftedSpecific(msg.sender, recipeId, R.outId, R.outAmount);
    }

    /**
     * @notice Ejecuta una receta general: quema `burnTotal` unidades con IDs/cantidades a elección del usuario.
     * @param recipeId        Id de la receta.
     * @param burnIds         Lista de IDs a quemar (cualquiera del Core).
     * @param burnAmounts     Cantidades por ID. Deben sumar >= burnTotal configurado.
     */
    function craftAny(
        uint256 recipeId,
        uint256[] calldata burnIds,
        uint256[] calldata burnAmounts
    ) external nonReentrant onlyWhenAuthorized {
        AnyRecipe memory R = anyRecipes[recipeId];
        require(R.active, "Recipe inactive");
        require(burnIds.length > 0, "Empty burnIds");
        require(burnIds.length == burnAmounts.length, "Length mismatch");

        uint256 total;
        for (uint256 i = 0; i < burnIds.length; i++) {
            uint256 amt = burnAmounts[i];
            require(amt > 0, "Burn amt=0");
            require(traitsCore.balanceOf(msg.sender, burnIds[i]) >= amt, "Insufficient balance for an id");
            total += amt;
        }
        require(total >= R.burnTotal, "Total burn below requirement");

        for (uint256 i = 0; i < burnIds.length; i++) {
            traitsCore.burnFromExtension(msg.sender, burnIds[i], burnAmounts[i]);
        }

        traitsCore.mintFromExtension(msg.sender, R.outId, R.outAmount);

        emit CraftedAny(msg.sender, recipeId, R.outId, R.outAmount, total);
    }

    /* ============ Lecturas ============ */

    function getSpecificRecipe(uint256 recipeId)
        external
        view
        returns (
            bool active,
            uint256[] memory burnIds,
            uint256[] memory burnAmounts,
            uint256 outId,
            uint256 outAmount
        )
    {
        SpecificRecipe storage R = specificRecipes[recipeId];
        return (R.active, R.burnIds, R.burnAmounts, R.outId, R.outAmount);
    }

    function getAnyRecipe(uint256 recipeId)
        external
        view
        returns (bool active, uint256 burnTotal, uint256 outId, uint256 outAmount)
    {
        AnyRecipe memory R = anyRecipes[recipeId];
        return (R.active, R.burnTotal, R.outId, R.outAmount);
    }

    /* ============ Admin helpers ============ */

    function setTraitsCore(address newCore) external onlyOwner {
        require(newCore != address(0), "zero addr");
        traitsCore = IAdrianTraitsCore(newCore);
    }
}
